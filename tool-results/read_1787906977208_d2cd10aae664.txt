     3→ * Material 3 Expressive — MCP Server
     4→ * ===================================
     5→ * Exposes the m3-expressive-react component library to AI coding agents
     6→ * via the Model Context Protocol, over TWO transports:
     7→ *
     8→ *   1. stdio (default)            — `bun start` / `bun index.ts`
     9→ *      Spawned BY the MCP client (Claude Code, Cursor, …). mcp.json config
    10→ *      in the README stays valid.
    11→ *
    12→ *   2. streamable HTTP (stateless)— `bun run dev` / `bun index.ts --http`
    13→ *      Long-running daemon on port 3210 (override with PORT env). Browser
    14→ *      clients and remote agents POST JSON-RPC to /mcp; CORS is open so the
    15→ *      showcase can connect cross-origin through the gateway.
    16→ *
    17→ * Connect (Claude Code / any MCP-capable client) — mcp.json:
    18→ *   {
    19→ *     "mcpServers": {
    20→ *       "m3-expressive": {
    21→ *         "command": "bun",
    22→ *         "args": ["run", "--cwd", "/ABSOLUTE/PATH/TO/PROJECT/mini-services/mcp-server", "start"]
    23→ *       }
    24→ *     }
    25→ *   }
    26→ *
    27→ * Tools surface (structured component knowledge, NOT raw file dumps):
    28→ *   list_components · get_component · get_component_api ·
    29→ *   get_component_examples · get_component_guidelines · get_component_states ·
    30→ *   get_component_source · search_components · list_themes · get_theme ·
    31→ *   generate_theme · get_design_tokens
    32→ *
    33→ * Resources (read-only, agentic deepening):
    34→ *   m3://handbook · m3://components · m3://components/{id} (template) ·
    35→ *   m3://tokens · m3://themes · m3://package

    37→ * Prompts (task playbooks that drive the tools):
    38→ *   m3_screen_builder · m3_style_audit · m3_theme_seed

   121→function search(query: string) {
   122→  const q = query.toLowerCase().trim();
   123→  if (!q) return METAS.map(summary);
   124→  return METAS.filter((m) => {
   125→    const haystack = [
   126→      m.id, m.name, m.category, m.description, m.variants?.join(" ") ?? "",
   127→      m.guidelines?.whenToUse?.join(" ") ?? "",
   128→      m.related?.join(" ") ?? "",
   129→      m.props.map((p) => `${p.name} ${p.description}`).join(" "),
   130→    ].join(" ").toLowerCase();
   131→    return q.split(/\s+/).every((term) => haystack.includes(term));
   132→  }).map(summary);
   133→}

   258→const text = (data: unknown) => ({ content: [{ type: "text" as const, text: JSON.stringify(data, null, 2) }] });

   260→/**
   261→ * Build a fully-configured McpServer (14 tools · 6 read-only resources · 3 prompts).
   262→ *
   263→ * Called ONCE in stdio mode and once per request in stateless HTTP mode —
   264→ * the SDK requires a fresh transport (and therefore a fresh server
   265→ * connection) per request when `sessionIdGenerator: undefined`.
   266→ */
   267→function buildServer(): McpServer {
   268→  const server = new McpServer({
   269→    name: "m3-expressive",
   270→    version: "1.0.0",
   271→  });
   272→
   273→  /* ---- discovery ---- */
   274→  server.registerTool(
   275→    "list_components",
   276→    {
   277→      title: "List components",
   278→      description:
   279→        "List every component in the Material 3 Expressive React library with id, name, category, variants, M3E flag, source path and import line. Start here.",
   280→      inputSchema: { category: z.string().optional().describe("Filter by category: actions | communication | containment | selection | textinput | navigation | feedback") },
   281→    },
   282→    ({ category }) => {
   283→      let list = METAS.map(summary);
   284→      if (category) {
   285→        if (!CATEGORIES.includes(category as (typeof CATEGORIES)[number])) {
   286→          return { content: [{ type: "text" as const, text: `Unknown category "${category}". Valid: ${CATEGORIES.join(", ")}` }], isError: true };
   287→        }
   288→        list = list.filter((c) => c.category === category);
   289→      }
   290→      return text({
   291→        library: "m3-expressive-react",
   292→        version: "1.0.0",
   293→        spec: "https://m3.material.io",
   294→        totalCount: list.length,
   295→        components: list,
   296→      });
   297→    }
   298→  );
   299→
   300→  server.registerTool(
   301→    "search_components",
   302→    {
   303→      title: "Search components",
   304→      description: "Full-text search across ids, names, descriptions, variants, when-to-use guidance and props. Empty query lists all.",
   305→      inputSchema: { query: z.string().describe("Space-separated keywords, e.g. 'floating label error'") },
   306→    },
   307→    ({ query }) => {
   308→      const results = search(query ?? "");
   309→      return text({ query: query ?? "", count: results.length, results });
   310→    }
   311→  );
   312→
   313→  server.registerTool(
   314→    "get_component",
   315→    {
   316→      title: "Get component",

   830→/* ------------------------------------------------------------------ */
   831→/* Transport selection                                                 */
   832→/* ------------------------------------------------------------------ */
   833→/** stdio is the default; HTTP via MCP_TRANSPORT=http (the `dev` script) or `--http`. */
   834→const HTTP_REQUESTED =
   835→  process.env.MCP_TRANSPORT === "http" || process.argv.includes("--http");
   836→
   837→if (!HTTP_REQUESTED) {
   838→  /* ---------------- stdio (default — MCP clients spawn this) --------------- */
   839→  await buildServer().connect(new StdioServerTransport());
   840→  console.error(
   841→    `[m3-expressive-mcp] connected on stdio — ${METAS.length} components, ${m3Themes.length} themes, full token system`
   842→  );
   843→} else {
   844→  /* ---------------- streamable HTTP (stateless) ---------------------------- */
   845→  const PORT = Number(process.env.PORT) || 3210;
   846→
   847→  const CORS_HEADERS: Record<string, string> = {
   848→    "Access-Control-Allow-Origin": "*",
   849→    "Access-Control-Allow-Methods": "GET, POST, DELETE, OPTIONS",
   850→    "Access-Control-Allow-Headers": "content-type, accept, authorization, mcp-session-id, mcp-protocol-version, last-event-id",
   851→    "Access-Control-Expose-Headers": "mcp-session-id, mcp-protocol-version",
   852→    "Access-Control-Max-Age": "86400",
   853→  };
   854→
   855→  const applyCors = (res: ServerResponse) => {
   856→    for (const [k, v] of Object.entries(CORS_HEADERS)) res.setHeader(k, v);
   857→  };
   858→
   859→  const sendJson = (res: ServerResponse, status: number, body: unknown) => {
   860→    res.statusCode = status;
   861→    res.setHeader("content-type", "application/json");
   862→    res.end(JSON.stringify(body));
   863→  };
   864→
   865→  const readBody = (req: IncomingMessage) =>
   866→    new Promise<string>((resolveBody, rejectBody) => {
   867→      const chunks: Buffer[] = [];
   868→      req.on("data", (c: Buffer) => chunks.push(c));
   869→      req.on("end", () => resolveBody(Buffer.concat(chunks).toString("utf8")));
   870→      req.on("error", rejectBody);
   871→    });
   872→
   873→  /**
   874→   * POST /mcp — JSON-RPC over streamable HTTP.
   875→   *
   876→   * Stateless mode (`sessionIdGenerator: undefined`): the SDK requires a
   877→   * fresh transport + server per request, so there is no Mcp-Session-Id and
   878→   * every POST is independent. `enableJsonResponse: true` answers plain
   879→   * application/json (no SSE stream) — ideal for curl and browser gateways.
   880→   */
   881→  async function handleMcpPost(req: IncomingMessage, res: ServerResponse) {
   882→    applyCors(res);
   883→
   884→    // Lenient header normalization so plain curl / browser fetch() work:
   885→    // the MCP spec requires Accept: application/json, text/event-stream.
   886→    const h = req.headers;
   887→    const accept = new Set(
   888→      String(h.accept ?? "").split(",").map((s) => s.trim()).filter(Boolean)
   889→    );
   890→    accept.add("application/json");
   891→    accept.add("text/event-stream");
   892→    h.accept = [...accept].join(", ");
   893→    if (!h["content-type"]) h["content-type"] = "application/json";
   894→
   895→    let parsed: unknown;
   896→    try {
   897→      parsed = JSON.parse(await readBody(req));
   898→    } catch {
   899→      return sendJson(res, 400, {
   900→        jsonrpc: "2.0", error: { code: -32700, message: "Parse error" }, id: null,
   901→      });
   902→    }
   903→
   904→    const transport = new StreamableHTTPServerTransport({
   905→      sessionIdGenerator: undefined, // stateless — no Mcp-Session-Id issued
   906→      enableJsonResponse: true,      // plain JSON responses (no SSE streams)
   907→    });
   908→    const server = buildServer();
   909→
   910→    // Release per-request state once the response connection closes.
   911→    res.on("close", () => {
   912→      void transport.close();
   913→      void server.close();
   914→    });
   915→
   916→    try {
   917→      await server.connect(transport);
   918→      await transport.handleRequest(req, res, parsed);
   919→    } catch (e) {
   920→      if (!res.headersSent) {
   921→        sendJson(res, 500, {
   922→          jsonrpc: "2.0", error: { code: -32603, message: `Internal error: ${e}` }, id: null,
   923→        });
   924→      }
   925→    }
   926→  }
   927→
   928→  const httpServer = createServer(async (req, res) => {
   929→    const url = new URL(req.url ?? "/", "http://localhost");
   930→
   931→    try {
   932→      if (url.pathname === "/") {
   933→        applyCors(res);
   934→        return sendJson(res, 200, {
   935→          service: "m3-expressive-mcp",
   936→          transport: "streamable-http",
   937→          version: "1.0.0",
   938→          tools: 14,
   939→          components: METAS.length,
   940→          protocol: "MCP streamable HTTP (stateless — no Mcp-Session-Id)",
   941→          endpoints: { health: "GET /", mcp: "POST /mcp" },
   942→          status: "ok",
   943→        });
   944→      }
   945→
   946→      if (url.pathname === "/mcp") {
   947→        switch (req.method) {
   948→          case "OPTIONS":
   949→            applyCors(res);
   950→            res.statusCode = 204;
   951→            return res.end();
   952→          case "POST":
   953→            return await handleMcpPost(req, res);
   954→          case "GET":
   955→            // Stateless mode has no server-initiated SSE stream → 405 per spec.
   956→            applyCors(res);
   957→            res.setHeader("Allow", "POST, OPTIONS");
   958→            return sendJson(res, 405, {
   959→              jsonrpc: "2.0",
   960→              error: { code: -32000, message: "Method not allowed. GET /mcp (SSE) is not supported in stateless mode; use POST." },
   961→              id: null,
   962→            });
   963→          case "DELETE":
   964→            applyCors(res);
   965→            res.statusCode = 204;
   966→            return res.end();
   967→          default:
   968→            applyCors(res);
   969→            res.setHeader("Allow", "POST, GET, DELETE, OPTIONS");
   970→            return sendJson(res, 405, {
   971→              jsonrpc: "2.0", error: { code: -32000, message: "Method not allowed." }, id: null,
   972→            });
   973→        }
   974→      }
   975→
   976→      sendJson(res, 404, { error: "Not found. Use GET / (health) or POST /mcp (JSON-RPC)." });
   977→    } catch (e) {
   978→      if (!res.headersSent) sendJson(res, 500, { error: `${e}` });
   979→      else res.end();
   980→    }
   981→  });
