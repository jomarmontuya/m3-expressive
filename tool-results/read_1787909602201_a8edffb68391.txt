    26→    26→function startOfDay(d: Date): Date {
    27→    27→  return new Date(d.getFullYear(), d.getMonth(), d.getDate());
    28→    28→}
    29→    29→
    30→    30→function startOfMonth(d: Date): Date {
    31→    31→  return new Date(d.getFullYear(), d.getMonth(), 1);

    34→    34→function sameDay(a: Date, b: Date): boolean {
    35→    35→  return (
    36→    36→    a.getFullYear() === b.getFullYear() &&
    37→    37→    a.getMonth() === b.getMonth() &&
    38→    38→    a.getDate() === b.getDate()
    39→    39→  );
    40→    40→}

    42→    42→function isoOf(d: Date): string {
    43→    43→  const mm = String(d.getMonth() + 1).padStart(2, "0");
    44→    44→  const dd = String(d.getDate()).padStart(2, "0");
    45→    45→  return `${d.getFullYear()}-${mm}-${dd}`;
    46→    46→}

    48→    48→/** 6×7 grid of days (weeks start on Sunday) covering the cursor month */
    49→    49→function getMonthGrid(cursor: Date): Date[] {
    50→    50→  const first = startOfMonth(cursor);
    51→    51→  const start = new Date(first.getFullYear(), first.getMonth(), 1 - first.getDay());
    52→    52→  const cells: Date[] = [];
    53→    53→  for (let i = 0; i < 42; i++) {
    54→    54→    cells.push(new Date(start.getFullYear(), start.getMonth(), start.getDate() + i));
    55→    55→  }
    56→    56→  return cells;
    57→    57→}

    69→    69→/** Official date-range shape (androidx DateRangePicker / MaterialDatePicker) */
    70→    70→export interface DateRange {
    71→    71→  start?: Date;
    72→    72→  end?: Date;
    73→    73→}
    74→    74→
    75→    75→/**
    76→    76→ * Range tap state machine (androidx DateRangePicker convention):
    77→    77→ * no start → tap sets start · start-only → tap ≥ start completes the range,
    78→    78→ * tap < start restarts with a new start · complete → tap restarts fresh.
    79→    79→ */
    80→    80→function advanceRange(current: DateRange, d: Date): DateRange {
    81→    81→  if (!current.start) return { start: d, end: undefined };
    82→    82→  if (!current.end) {
    83→    83→    if (startOfDay(d) >= startOfDay(current.start)) return { start: current.start, end: d };
    84→    84→    return { start: d, end: undefined };
    85→    85→  }
    86→    86→  return { start: d, end: undefined };
    87→    87→}

    96→    96→interface DatePickerCalendarProps {
    97→    97→  value?: Date;
    98→    98→  onChange?: (d: Date) => void;
    99→    99→  minDate?: Date;
   100→   100→  maxDate?: Date;
   101→   101→  /**
   102→   102→   * Selected-day container color. Inline uses the library's primary pill;
   103→   103→   * the modal uses the official androidx DatePickerTokens
   104→   104→   * SelectedDateContainerColor = primary-container.
   105→   105→   */
   106→   106→  tone?: "primary" | "primary-container";
   107→   107→  /** Slide/fade micro-transition on the month-year label (modal only). */
   108→   108→  animatedHeader?: boolean;
   109→   109→  /** 'single' (default) picks one date; 'range' picks a start/end pair. */
   110→   110→  selectionMode?: "single" | "range";
   111→   111→  /** Range mode — selected range; uncontrolled when omitted. */
   112→   112→  range?: DateRange;
   113→   113→  /** Range mode — fires on every tap with the next range (partial included). */
   114→   114→  onRangeChange?: (r: DateRange) => void;
   115→   115→}

   117→   117→/**
   118→   118→ * Header row + month grid + year grid, shared verbatim by the inline and
   119→   119→ * modal presentations. No outer panel/padding — the caller supplies the
   120→   120→ * surface. Owns its view state (cursor, month/year, roving tabindex), so
   121→   121→ * each mount starts on the selected/today month.
   122→   122→ */
   123→   123→function DatePickerCalendar({
   124→   124→  value,
   125→   125→  onChange,
   126→   126→  minDate,
   127→   127→  maxDate,
   128→   128→  tone = "primary",
   129→   129→  animatedHeader = false,
   130→   130→  selectionMode = "single",
   131→   131→  range,
   132→   132→  onRangeChange,
   133→   133→}: DatePickerCalendarProps) {
   134→   134→  const [internal, setInternal] = React.useState<Date | undefined>(undefined);
   135→   135→  const selected = value ?? internal;
   136→   136→  const [internalRange, setInternalRange] = React.useState<DateRange>({});
   137→   137→  const rangeOn = selectionMode === "range";
   138→   138→  const rangeSel = rangeOn ? (range ?? internalRange) : undefined;
   139→   139→  // Hover/focus preview source while only the start of the range is set
   140→   140→  const [hoverDate, setHoverDate] = React.useState<Date | undefined>(undefined);
   141→   141→  const previewFrom =
   142→   142→    rangeOn && rangeSel?.start !== undefined && rangeSel?.end === undefined
   143→   143→      ? rangeSel.start
   144→   144→      : undefined;
   145→   145→  const [cursor, setCursor] = React.useState<Date>(() => startOfMonth(value ?? new Date()));
   146→   146→  const [view, setView] = React.useState<"month" | "year">("month");
   147→   147→  const [rovingOverride, setRovingOverride] = React.useState<string | null>(null);
   148→   148→  // Scope the shared selection pill so multiple pickers never cross-animate
   149→   149→  const pillId = React.useId();
   150→   150→
   151→   151→  const selectedYearRef = React.useRef<HTMLButtonElement | null>(null);
   152→   152→  const gridRef = React.useRef<HTMLDivElement | null>(null);
   153→   153→
   154→   154→  React.useEffect(() => {
   155→   155→    if (view === "year") selectedYearRef.current?.scrollIntoView({ block: "center" });
   156→   156→  }, [view]);
   157→   157→
   158→   158→  React.useEffect(() => {
   159→   159→    setRovingOverride(null);
   160→   160→  }, [cursor]);
   161→   161→
   162→   162→  const cells = React.useMemo(() => getMonthGrid(cursor), [cursor]);
   163→   163→
   164→   164→  const years = React.useMemo(() => {
   165→   165→    const list: number[] = [];
   166→   166→    const last = new Date().getFullYear() + 10;
   167→   167→    for (let y = FIRST_YEAR; y <= last; y++) list.push(y);
   168→   168→    return list;
   169→   169→  }, []);
   170→   170→
   171→   171→  const isDisabledDay = React.useCallback(
   172→   172→    (d: Date) => {
   173→   173→      if (minDate && startOfDay(d) < startOfDay(minDate)) return true;
   174→   174→      if (maxDate && startOfDay(d) > startOfDay(maxDate)) return true;
   175→   175→      return false;
   176→   176→    },
   177→   177→    [minDate, maxDate]
   178→   178→  );
   179→   179→
   180→   180→  const handleSelect = (d: Date) => {
   181→   181→    if (rangeOn) {
   182→   182→      const next = advanceRange(rangeSel ?? {}, d);
   183→   183→      setInternalRange(next);
   184→   184→      onRangeChange?.(next);
   185→   185→      return;
   186→   186→    }
   187→   187→    setInternal(d);
   188→   188→    onChange?.(d);
   189→   189→  };
   190→   190→
   191→   191→  const navigate = (dir: number) => {
   192→   192→    setCursor((c) =>
   193→   193→      view === "year"
   194→   194→        ? new Date(c.getFullYear() + dir * 12, c.getMonth(), 1)
   195→   195→        : new Date(c.getFullYear(), c.getMonth() + dir, 1)
   196→   196→    );
   197→   197→  };
   198→   198→
   199→   199→  const today = new Date();
   200→   200→  // Range mode anchors on the range start (the first picked day)
   201→   201→  const anchor = rangeOn ? rangeSel?.start : selected;
   202→   202→  const highlightYear = (anchor ?? cursor).getFullYear();
   203→   203→
   204→   204→  // Roving tabindex anchor: selected day in view → today → first enabled day of month.
   205→   205→  const activeIso = React.useMemo(() => {
   206→   206→    if (rovingOverride) return rovingOverride;
   207→   207→    const inView = (d: Date) => cells.some((c) => sameDay(c, d));
   208→   208→    const pick =
   209→   209→      (anchor && inView(anchor) && !isDisabledDay(anchor) ? anchor : undefined) ??
   210→   210→      (inView(today) && !isDisabledDay(today) ? today : undefined) ??
   211→   211→      cells.find((c) => c.getMonth() === cursor.getMonth() && !isDisabledDay(c)) ??
   212→   212→      cells.find((c) => !isDisabledDay(c)) ??
   213→   213→      cells[0];
   214→   214→    return isoOf(pick);
   215→   215→  }, [rovingOverride, cells, anchor, today, cursor, isDisabledDay]);
   216→   216→
   217→   217→  const focusCell = (iso: string) => {
   218→   218→    requestAnimationFrame(() => {
   219→   219→      gridRef.current
   220→   220→        ?.querySelector<HTMLButtonElement>(`button[data-iso="${iso}"]`)
   221→   221→        ?.focus();
   222→   222→    });
   223→   223→  };
   224→   224→
   225→   225→  /** Arrow-key day navigation: ←/→ ±1 day, ↑/↓ ±7, Home/End week bounds */
   226→   226→  const handleDayKeyDown = (e: React.KeyboardEvent<HTMLButtonElement>, idx: number) => {
   227→   227→    let delta = 0;
   228→   228→    if (e.key === "ArrowLeft") delta = -1;
   229→   229→    else if (e.key === "ArrowRight") delta = 1;
   230→   230→    else if (e.key === "ArrowUp") delta = -7;
   231→   231→    else if (e.key === "ArrowDown") delta = 7;
   232→   232→    else if (e.key === "Home") delta = -(idx % 7);
   233→   233→    else if (e.key === "End") delta = 6 - (idx % 7);
   234→   234→    else return;
   235→   235→    e.preventDefault();
   236→   236→    let i = idx + delta;
   237→   237→    while (i >= 0 && i < cells.length && isDisabledDay(cells[i])) i += delta;
   238→   238→    if (i < 0 || i >= cells.length) return;
   239→   239→    const target = cells[i];
   240→   240→    if (target.getMonth() !== cursor.getMonth() || target.getFullYear() !== cursor.getFullYear()) {
   241→   241→      setCursor(startOfMonth(target));
   242→   242→    }
   243→   243→    setRovingOverride(isoOf(target));
   244→   244→    focusCell(isoOf(target));
   245→   245→  };
   246→   246→
   247→   247→  const weekRows: Date[][] = [];
   248→   248→  for (let r = 0; r < 6; r++) weekRows.push(cells.slice(r * 7, r * 7 + 7));
   249→   249→
   250→   250→  const headerLabel =
   251→   251→    view === "year"
   252→   252→      ? String(cursor.getFullYear())
   253→   253→      : `${MONTH_NAMES[cursor.getMonth()]} ${cursor.getFullYear()}`;
   254→   254→
   255→   255→  const selectedPillClass =
   256→   256→    tone === "primary-container" ? "bg-m3-primary-container" : "bg-m3-primary";
   257→   257→  const selectedTextClass =
   258→   258→    tone === "primary-container" ? "text-m3-on-primary-container" : "text-m3-on-primary";
   259→   259→
   260→   260→  return (
   261→   261→    <>
   262→   262→      {/* Header: month-year label + month navigation */}
   263→   263→      <div className="mb-1 flex items-center justify-between">
   264→   264→        <button
   265→   265→          type="button"
   266→   266→          onClick={() => setView((v) => (v === "month" ? "year" : "month"))}
   267→   267→          className="m3-state m3-focus md-title-large cursor-pointer rounded-full px-3 py-1 text-m3-on-surface outline-none"
   268→   268→        >
   269→   269→          {animatedHeader ? (
   270→   270→            <AnimatePresence mode="popLayout" initial={false}>
   271→   271→              <motion.span
   272→   272→                key={headerLabel}
   273→   273→                initial={{ opacity: 0, y: 6 }}
   274→   274→                animate={{ opacity: 1, y: 0 }}
   275→   275→                exit={{ opacity: 0, y: -6 }}
   276→   276→                transition={springs.fastVisual}
   277→   277→                className="block"
   278→   278→              >
   279→   279→                {headerLabel}
   280→   280→              </motion.span>
   281→   281→            </AnimatePresence>
   282→   282→          ) : (
   283→   283→            headerLabel
   284→   284→          )}
   285→   285→        </button>
   286→   286→        <div className="flex items-center">
   287→   287→          <button
   288→   288→            type="button"
   289→   289→            aria-label={view === "year" ? "Previous years" : "Previous month"}
   290→   290→            onClick={() => navigate(-1)}
   291→   291→            className="m3-state m3-focus flex h-12 w-12 cursor-pointer items-center justify-center rounded-full text-m3-on-surface outline-none"
   292→   292→          >
   293→   293→            <MaterialSymbol icon="chevron_left" />
   294→   294→          </button>
   295→   295→          <button
   296→   296→            type="button"
   297→   297→            aria-label={view === "year" ? "Next years" : "Next month"}
   298→   298→            onClick={() => navigate(1)}
   299→   299→            className="m3-state m3-focus flex h-12 w-12 cursor-pointer items-center justify-center rounded-full text-m3-on-surface outline-none"
   300→   300→          >
   301→   301→            <MaterialSymbol icon="chevron_right" />
   302→   302→          </button>
   303→   303→        </div>
   304→   304→      </div>
   305→   305→
   306→   306→      {view === "month" ? (
   307→   307→        <div role="grid" ref={gridRef}>
   308→   308→          {/* Weekday row */}
   309→   309→          <div role="row" className="grid grid-cols-7 justify-items-center">
   310→   310→            {WEEKDAYS.map((w, i) => (
   311→   311→              <span
   312→   312→                key={`${w.long}-${i}`}
   313→   313→                role="columnheader"
   314→   314→                aria-label={w.long}
   315→   315→                className="flex h-10 w-10 items-center justify-center md-label-medium text-m3-on-surface-variant"
   316→   316→              >
   317→   317→                {w.initial}
   318→   318→              </span>
   319→   319→            ))}
   320→   320→          </div>
   321→   321→          {/* Day grid — 40dp circular cells, roving tabindex + arrow keys */}
   322→   322→          {weekRows.map((week, r) => (
   323→   323→            <div key={`week-${r}`} role="row" className="grid grid-cols-7 justify-items-center">
   324→   324→              {week.map((day, c) => {
   325→   325→                const iso = isoOf(day);
   326→   326→                const idx = r * 7 + c;
   327→   327→                const isSelected = !rangeOn && selected !== undefined && sameDay(day, selected);
   328→   328→                const isToday = sameDay(day, today);
   329→   329→                const inMonth = day.getMonth() === cursor.getMonth();
   330→   330→                const disabled = isDisabledDay(day);
   331→   331→                // Range mode: committed membership
   332→   332→                const isRangeStart = rangeOn && rangeSel?.start !== undefined && sameDay(day, rangeSel.start);
   333→   333→                const isRangeEnd = rangeOn && rangeSel?.end !== undefined && sameDay(day, rangeSel.end);
   334→   334→                const inCommittedRange =
   335→   335→                  rangeOn &&
   336→   336→                  rangeSel?.start !== undefined &&
   337→   337→                  rangeSel?.end !== undefined &&
   338→   338→                  startOfDay(day) > startOfDay(rangeSel.start) &&
   339→   339→                  startOfDay(day) < startOfDay(rangeSel.end);
   340→   340→                // Tentative preview while only the start is set: hovering/focusing a
   341→   341→                // later day paints the band start→hover, an earlier day previews a restart.
   342→   342→                const previewEnd =
   343→   343→                  previewFrom !== undefined &&
   344→   344→                  hoverDate !== undefined &&
   345→   345→                  sameDay(day, hoverDate) &&
   346→   346→                  startOfDay(day) > startOfDay(previewFrom);
   347→   347→                const inPreviewRange =
   348→   348→                  previewFrom !== undefined &&
   349→   349→                  hoverDate !== undefined &&
   350→   350→                  startOfDay(day) > startOfDay(previewFrom) &&
   351→   351→                  startOfDay(day) < startOfDay(hoverDate);
   352→   352→                const previewRestart =
   353→   353→                  previewFrom !== undefined &&
   354→   354→                  hoverDate !== undefined &&
   355→   355→                  sameDay(day, hoverDate) &&
   356→   356→                  startOfDay(day) < startOfDay(previewFrom);
   357→   357→                // Band segment per cell: start = right half (rounded-l, hidden under the
   358→   358→                // circle), end = left half (rounded-r), in-between = full & square;
   359→   359→                // square cuts at week-row edges, 4dp vertical inset (inset-y-1).
   360→   360→                const bandKind: string | null =
   361→   361→                  isRangeStart && !isRangeEnd ? "start"
   362→   362→                  : isRangeEnd && !isRangeStart ? "end"
   363→   363→                  : inCommittedRange ? "mid"
   364→   364→                  : previewEnd ? "preview-end"
   365→   365→                  : inPreviewRange ? "preview-mid"
   366→   366→                  : null;
   367→   367→                const rangeProps =
   368→   368→                  rangeOn && previewFrom !== undefined
   369→   369→                    ? {
   370→   370→                        onMouseEnter: () => setHoverDate(day),
   371→   371→                        onMouseLeave: () => setHoverDate(undefined),
   372→   372→                        onFocus: () => setHoverDate(day),
   373→   373→                        onBlur: () => setHoverDate(undefined),
   374→   374→                      }
   375→   375→                    : {};
   376→   376→                return (
   377→   377→                  <div
   378→   378→                    key={iso}
   379→   379→                    role="gridcell"
   380→   380→                    aria-selected={
   381→   381→                      isSelected || isRangeStart || isRangeEnd || inCommittedRange || undefined
   382→   382→                    }
   383→   383→                    aria-current={isToday ? "date" : undefined}
   384→   384→                    aria-disabled={disabled || undefined}
   385→   385→                    className="relative flex w-full items-center justify-center"
   386→   386→                  >
   387→   387→                    {bandKind && (
   388→   388→                      <span
   389→   389→                        aria-hidden="true"
   390→   390→                        className={cn(
   391→   391→                          "pointer-events-none absolute inset-y-1",
   392→   392→                          bandKind === "start" &&
   393→   393→                            "left-1/2 right-0 rounded-l-full bg-m3-primary-container/44",
   394→   394→                          bandKind === "end" &&
   395→   395→                            "left-0 right-1/2 rounded-r-full bg-m3-primary-container/44",
   396→   396→                          bandKind === "mid" && "inset-x-0 bg-m3-primary-container/44",
   397→   397→                          bandKind === "preview-end" &&
   398→   398→                            "left-0 right-1/2 rounded-r-full bg-m3-primary-container/24",
   399→   399→                          bandKind === "preview-mid" && "inset-x-0 bg-m3-primary-container/24"
   400→   400→                        )}
   401→   401→                      />
   402→   402→                    )}
   403→   403→                    <button
   404→   404→                      type="button"
   405→   405→                      data-iso={iso}
   406→   406→                      disabled={disabled}
   407→   407→                      tabIndex={iso === activeIso ? 0 : -1}
   408→   408→                      aria-label={`${MONTH_NAMES[day.getMonth()]} ${day.getDate()}, ${day.getFullYear()}${
   409→   409→                        isRangeStart ? ", start of range" : isRangeEnd ? ", end of range" : ""
   410→   410→                      }`}
   411→   411→                      onClick={() => handleSelect(day)}
   412→   412→                      onKeyDown={(e) => handleDayKeyDown(e, idx)}
   413→   413→                      {...rangeProps}
   414→   414→                      className={cn(
   415→   415→                        "m3-state m3-focus relative my-0.5 flex h-10 w-10 cursor-pointer items-center justify-center rounded-full md-body-large outline-none",
   416→   416→                        disabled && "pointer-events-none opacity-38",
   417→   417→                        !isSelected && isToday && "border border-m3-primary text-m3-primary",
   418→   418→                        !isSelected && (previewEnd || previewRestart) && "border border-m3-primary",
   419→   419→                        !isSelected &&
   420→   420→                          !isToday &&
   421→   421→                          !previewEnd &&
   422→   422→                          !previewRestart &&
   423→   423→                          (inMonth ? "text-m3-on-surface" : "text-m3-on-surface-variant")
   424→   424→                      )}
   425→   425→                    >
   426→   426→                      {rangeOn ? (
   427→   427→                        isRangeStart && (
   428→   428→                          <motion.span
   429→   429→                            layoutId={`${pillId}-start`}
   430→   430→                            className={cn("absolute inset-0 rounded-full", selectedPillClass)}
   431→   431→                            transition={springs.expressive}
   432→   432→                          />
   433→   433→                        )
   434→   434→                      ) : (
   435→   435→                        isSelected && (
   436→   436→                          <motion.span
   437→   437→                            layoutId={pillId}
   438→   438→                            className={cn("absolute inset-0 rounded-full", selectedPillClass)}
   439→   439→                            transition={springs.expressive}
   440→   440→                          />
   441→   441→                        )
   442→   442→                      )}
   443→   443→                      {rangeOn && isRangeEnd && !isRangeStart && (
   444→   444→                        <motion.span
