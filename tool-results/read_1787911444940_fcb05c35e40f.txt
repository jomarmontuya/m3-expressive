    14→const WEEKDAYS = [
    15→  { initial: "S", long: "Sunday" },
    16→  { initial: "M", long: "Monday" },

    24→const FIRST_YEAR = 1988;
    25→
    26→function startOfDay(d: Date): Date {
    27→  return new Date(d.getFullYear(), d.getMonth(), d.getDate());
    28→}
    29→
    30→function startOfMonth(d: Date): Date {
    31→  return new Date(d.getFullYear(), d.getMonth(), 1);
    32→}
    33→
    34→function sameDay(a: Date, b: Date): boolean {
    35→  return (
    36→    a.getFullYear() === b.getFullYear() &&
    37→    a.getMonth() === b.getMonth() &&
    38→    a.getDate() === b.getDate()
    39→  );
    40→}
    41→
    42→function isoOf(d: Date): string {
    43→  const mm = String(d.getMonth() + 1).padStart(2, "0");
    44→  const dd = String(d.getDate()).padStart(2, "0");
    45→  return `${d.getFullYear()}-${mm}-${dd}`;
    46→}
    47→
    48→/** 6×7 grid of days (weeks start on Sunday) covering the cursor month */
    49→function getMonthGrid(cursor: Date): Date[] {
    50→  const first = startOfMonth(cursor);
    51→  const start = new Date(first.getFullYear(), first.getMonth(), 1 - first.getDay());
    52→  const cells: Date[] = [];
    53→  for (let i = 0; i < 42; i++) {
    54→    cells.push(new Date(start.getFullYear(), start.getMonth(), start.getDate() + i));
    55→  }
    56→  return cells;
    57→}

    59→/** Modal header headline — "Fri, Aug 21" per the official selected-date header */
    60→function formatHeadline(d: Date): string {
    61→  return `${WEEKDAYS[d.getDay()].long.slice(0, 3)}, ${MONTH_NAMES[d.getMonth()].slice(0, 3)} ${d.getDate()}`;
    62→}
    63→
    64→/** "Aug 21" — range-pair format for the modal header (month short + day) */
    65→function formatShort(d: Date): string {
    66→  return `${MONTH_NAMES[d.getMonth()].slice(0, 3)} ${d.getDate()}`;
    67→}

    69→/** Official date-range shape (androidx DateRangePicker / MaterialDatePicker) */
    70→export interface DateRange {
    71→  start?: Date;
    72→  end?: Date;
    73→}
    74→
    75→/**
    76→ * Range tap state machine (androidx DateRangePicker convention):
    77→ * no start → tap sets start · start-only → tap ≥ start completes the range,
    78→ * tap < start restarts with a new start · complete → tap restarts fresh.
    79→ */
    80→function advanceRange(current: DateRange, d: Date): DateRange {
    81→  if (!current.start) return { start: d, end: undefined };
    82→  if (!current.end) {
    83→    if (startOfDay(d) >= startOfDay(current.start)) return { start: current.start, end: d };
    84→    return { start: d, end: undefined };
    85→  }
    86→  return { start: d, end: undefined };
    87→}

    96→interface DatePickerCalendarProps {
    97→  value?: Date;
    98→  onChange?: (d: Date) => void;
    99→  minDate?: Date;
   100→  maxDate?: Date;
   101→  /**
   102→   * Selected-day container color. Inline uses the library's primary pill;
   103→   * the modal uses the official androidx DatePickerTokens
   104→   * SelectedDateContainerColor = primary-container.
   105→   */
   106→  tone?: "primary" | "primary-container";
   107→  /** Slide/fade micro-transition on the month-year label (modal only). */
   108→  animatedHeader?: boolean;
   109→  /** 'single' (default) picks one date; 'range' picks a start/end pair. */
   110→  selectionMode?: "single" | "range";
   111→  /** Range mode — selected range; uncontrolled when omitted. */
   112→  range?: DateRange;
   113→  /** Range mode — fires on every tap with the next range (partial included). */
   114→  onRangeChange?: (r: DateRange) => void;
   115→}

   117→/**
   118→ * Header row + month grid + year grid, shared verbatim by the inline and
   119→ * modal presentations. No outer panel/padding — the caller supplies the
   120→ * surface. Owns its view state (cursor, month/year, roving tabindex), so
   121→ * each mount starts on the selected/today month.
   122→ */
   123→function DatePickerCalendar({
   124→  value,
   125→  onChange,
   126→  minDate,
   127→  maxDate,
   128→  tone = "primary",
   129→  animatedHeader = false,
   130→  selectionMode = "single",
   131→  range,
   132→  onRangeChange,
   133→}: DatePickerCalendarProps) {
   134→  const [internal, setInternal] = React.useState<Date | undefined>(undefined);
   135→  const selected = value ?? internal;
   136→  const [internalRange, setInternalRange] = React.useState<DateRange>({});
   137→  const rangeOn = selectionMode === "range";
   138→  const rangeSel = rangeOn ? (range ?? internalRange) : undefined;
   139→  // Hover/focus preview source while only the start of the range is set
   140→  const [hoverDate, setHoverDate] = React.useState<Date | undefined>(undefined);
   141→  const previewFrom =
   142→    rangeOn && rangeSel?.start !== undefined && rangeSel?.end === undefined
   143→      ? rangeSel.start
   144→      : undefined;
   145→  const [cursor, setCursor] = React.useState<Date>(() => startOfMonth(value ?? new Date()));
   146→  const [view, setView] = React.useState<"month" | "year">("month");
   147→  const [rovingOverride, setRovingOverride] = React.useState<string | null>(null);
   148→  // Scope the shared selection pill so multiple pickers never cross-animate
   149→  const pillId = React.useId();
   150→
   151→  const selectedYearRef = React.useRef<HTMLButtonElement | null>(null);
   152→  const gridRef = React.useRef<HTMLDivElement | null>(null);
   153→
   154→  React.useEffect(() => {
   155→    if (view === "year") selectedYearRef.current?.scrollIntoView({ block: "center" });
   156→  }, [view]);
   157→
   158→  React.useEffect(() => {
   159→    setRovingOverride(null);
   160→  }, [cursor]);
   161→
   162→  const cells = React.useMemo(() => getMonthGrid(cursor), [cursor]);
   163→
   164→  const years = React.useMemo(() => {
   165→    const list: number[] = [];
   166→    const last = new Date().getFullYear() + 10;
   167→    for (let y = FIRST_YEAR; y <= last; y++) list.push(y);
   168→    return list;
   169→  }, []);
   170→
   171→  const isDisabledDay = React.useCallback(
   172→    (d: Date) => {
   173→      if (minDate && startOfDay(d) < startOfDay(minDate)) return true;
   174→      if (maxDate && startOfDay(d) > startOfDay(maxDate)) return true;
   175→      return false;
   176→    },
   177→    [minDate, maxDate]
   178→  );
   179→
   180→  const handleSelect = (d: Date) => {
   181→    if (rangeOn) {
   182→      const next = advanceRange(rangeSel ?? {}, d);
   183→      setInternalRange(next);
   184→      onRangeChange?.(next);
   185→      return;
   186→    }
   187→    setInternal(d);
   188→    onChange?.(d);
   189→  };
   190→
   191→  const navigate = (dir: number) => {
   192→    setCursor((c) =>
   193→      view === "year"
   194→        ? new Date(c.getFullYear() + dir * 12, c.getMonth(), 1)
   195→        : new Date(c.getFullYear(), c.getMonth() + dir, 1)
   196→    );
   197→  };
   198→
   199→  const today = new Date();
   200→  // Range mode anchors on the range start (the first picked day)
   201→  const anchor = rangeOn ? rangeSel?.start : selected;
   202→  const highlightYear = (anchor ?? cursor).getFullYear();
   203→
   204→  // Roving tabindex anchor: selected day in view → today → first enabled day of month.
   205→  const activeIso = React.useMemo(() => {
   206→    if (rovingOverride) return rovingOverride;
   207→    const inView = (d: Date) => cells.some((c) => sameDay(c, d));
   208→    const pick =
   209→      (anchor && inView(anchor) && !isDisabledDay(anchor) ? anchor : undefined) ??
   210→      (inView(today) && !isDisabledDay(today) ? today : undefined) ??
   211→      cells.find((c) => c.getMonth() === cursor.getMonth() && !isDisabledDay(c)) ??
   212→      cells.find((c) => !isDisabledDay(c)) ??
   213→      cells[0];
   214→    return isoOf(pick);
   215→  }, [rovingOverride, cells, anchor, today, cursor, isDisabledDay]);
   216→
   217→  const focusCell = (iso: string) => {
   218→    requestAnimationFrame(() => {
   219→      gridRef.current
   220→        ?.querySelector<HTMLButtonElement>(`button[data-iso="${iso}"]`)
   221→        ?.focus();
   222→    });
   223→  };
   224→
   225→  /** Arrow-key day navigation: ←/→ ±1 day, ↑/↓ ±7, Home/End week bounds */
   226→  const handleDayKeyDown = (e: React.KeyboardEvent<HTMLButtonElement>, idx: number) => {
   227→    let delta = 0;
   228→    if (e.key === "ArrowLeft") delta = -1;
   229→    else if (e.key === "ArrowRight") delta = 1;
   230→    else if (e.key === "ArrowUp") delta = -7;
   231→    else if (e.key === "ArrowDown") delta = 7;
   232→    else if (e.key === "Home") delta = -(idx % 7);
   233→    else if (e.key === "End") delta = 6 - (idx % 7);
   234→    else return;
   235→    e.preventDefault();
   236→    let i = idx + delta;
   237→    while (i >= 0 && i < cells.length && isDisabledDay(cells[i])) i += delta;
   238→    if (i < 0 || i >= cells.length) return;
   239→    const target = cells[i];
   240→    if (target.getMonth() !== cursor.getMonth() || target.getFullYear() !== cursor.getFullYear()) {
   241→      setCursor(startOfMonth(target));
   242→    }
   243→    setRovingOverride(isoOf(target));
   244→    focusCell(isoOf(target));
   245→  };
   246→
   247→  const weekRows: Date[][] = [];
   248→  for (let r = 0; r < 6; r++) weekRows.push(cells.slice(r * 7, r * 7 + 7));
   249→
   250→  const headerLabel =
   251→    view === "year"
   252→      ? String(cursor.getFullYear())
   253→      : `${MONTH_NAMES[cursor.getMonth()]} ${cursor.getFullYear()}`;
   254→
   255→  const selectedPillClass =
   256→    tone === "primary-container" ? "bg-m3-primary-container" : "bg-m3-primary";
   257→  const selectedTextClass =
   258→    tone === "primary-container" ? "text-m3-on-primary-container" : "text-m3-on-primary";
   259→
   260→  return (
   261→    <>
   262→      {/* Header: month-year label + month navigation */}
   263→      <div className="mb-1 flex items-center justify-between">
   264→        <button
   265→          type="button"
   266→          onClick={() => setView((v) => (v === "month" ? "year" : "month"))}
   267→          className="m3-state m3-focus md-title-large cursor-pointer rounded-full px-3 py-1 text-m3-on-surface outline-none"
   268→        >
   269→          {animatedHeader ? (
   270→            <AnimatePresence mode="popLayout" initial={false}>
   271→              <motion.span
   272→                key={headerLabel}
   273→                initial={{ opacity: 0, y: 6 }}
   274→                animate={{ opacity: 1, y: 0 }}
   275→                exit={{ opacity: 0, y: -6 }}
   276→                transition={springs.fastVisual}
   277→                className="block"
   278→              >
   279→                {headerLabel}
   280→              </motion.span>
   281→            </AnimatePresence>
   282→          ) : (
   283→            headerLabel
   284→          )}
   285→        </button>
   286→        <div className="flex items-center">
   287→          <button
   288→            type="button"
   289→            aria-label={view === "year" ? "Previous years" : "Previous month"}
   290→            onClick={() => navigate(-1)}
   291→            className="m3-state m3-focus flex h-12 w-12 cursor-pointer items-center justify-center rounded-full text-m3-on-surface outline-none"
   292→          >
   293→            <MaterialSymbol icon="chevron_left" />
   294→          </button>
   295→          <button
   296→            type="button"
   297→            aria-label={view === "year" ? "Next years" : "Next month"}
   298→            onClick={() => navigate(1)}
   299→            className="m3-state m3-focus flex h-12 w-12 cursor-pointer items-center justify-center rounded-full text-m3-on-surface outline-none"
   300→          >
   301→            <MaterialSymbol icon="chevron_right" />
   302→          </button>
   303→        </div>
   304→      </div>
   305→
   306→      {view === "month" ? (
   307→        <div role="grid" ref={gridRef}>
   308→          {/* Weekday row */}
   309→          <div role="row" className="grid grid-cols-7 justify-items-center">
   310→            {WEEKDAYS.map((w, i) => (
   311→              <span
   312→                key={`${w.long}-${i}`}
   313→                role="columnheader"
   314→                aria-label={w.long}
   315→                className="flex h-10 w-10 items-center justify-center md-label-medium text-m3-on-surface-variant"
   316→              >
   317→                {w.initial}
   318→              </span>
   319→            ))}
   320→          </div>
   321→          {/* Day grid — 40dp circular cells, roving tabindex + arrow keys */}
   322→          {weekRows.map((week, r) => (
   323→            <div key={`week-${r}`} role="row" className="grid grid-cols-7 justify-items-center">
   324→              {week.map((day, c) => {
   325→                const iso = isoOf(day);
   326→                const idx = r * 7 + c;
   327→                const isSelected = !rangeOn && selected !== undefined && sameDay(day, selected);
   328→                const isToday = sameDay(day, today);
   329→                const inMonth = day.getMonth() === cursor.getMonth();
   330→                const disabled = isDisabledDay(day);
   331→                // Range mode: committed membership
   332→                const isRangeStart = rangeOn && rangeSel?.start !== undefined && sameDay(day, rangeSel.start);
   333→                const isRangeEnd = rangeOn && rangeSel?.end !== undefined && sameDay(day, rangeSel.end);
   334→                const inCommittedRange =
   335→                  rangeOn &&
   336→                  rangeSel?.start !== undefined &&
   337→                  rangeSel?.end !== undefined &&
   338→                  startOfDay(day) > startOfDay(rangeSel.start) &&
   339→                  startOfDay(day) < startOfDay(rangeSel.end);
   340→                // Tentative preview while only the start is set: hovering/focusing a
   341→                // later day paints the band start→hover, an earlier day previews a restart.
   342→                const previewEnd =
   343→                  previewFrom !== undefined &&
   344→                  hoverDate !== undefined &&
   345→                  sameDay(day, hoverDate) &&
   346→                  startOfDay(day) > startOfDay(previewFrom);
   347→                const inPreviewRange =
   348→                  previewFrom !== undefined &&
   349→                  hoverDate !== undefined &&
   350→                  startOfDay(day) > startOfDay(previewFrom) &&
   351→                  startOfDay(day) < startOfDay(hoverDate);
   352→                const previewRestart =
   353→                  previewFrom !== undefined &&
   354→                  hoverDate !== undefined &&
   355→                  sameDay(day, hoverDate) &&
   356→                  startOfDay(day) < startOfDay(previewFrom);
   357→                // Band segment per cell: start = right half (rounded-l, hidden under the
   358→                // circle), end = left half (rounded-r), in-between = full & square;
   359→                // square cuts at week-row edges, 4dp vertical inset (inset-y-1).
   360→                const bandKind: string | null =
   361→                  isRangeStart && !isRangeEnd ? "start"
   362→                  : isRangeEnd && !isRangeStart ? "end"
   363→                  : inCommittedRange ? "mid"
   364→                  : previewEnd ? "preview-end"
   365→                  : inPreviewRange ? "preview-mid"
   366→                  : null;
   367→                const rangeProps =
   368→                  rangeOn && previewFrom !== undefined
   369→                    ? {
   370→                        onMouseEnter: () => setHoverDate(day),
   371→                        onMouseLeave: () => setHoverDate(undefined),
   372→                        onFocus: () => setHoverDate(day),
   373→                        onBlur: () => setHoverDate(undefined),
   374→                      }
   375→                    : {};
   376→                return (
   377→                  <div
   378→                    key={iso}
   379→                    role="gridcell"
   380→                    aria-selected={
   381→                      isSelected || isRangeStart || isRangeEnd || inCommittedRange || undefined
   382→                    }
   383→                    aria-current={isToday ? "date" : undefined}
   384→                    aria-disabled={disabled || undefined}
   385→                    className="relative flex w-full items-center justify-center"
   386→                  >
   387→                    {bandKind && (
   388→                      <span
   389→                        aria-hidden="true"
   390→                        className={cn(
   391→                          "pointer-events-none absolute inset-y-1",
   392→                          bandKind === "start" &&
   393→                            "left-1/2 right-0 rounded-l-full bg-m3-primary-container/44",
   394→                          bandKind === "end" &&
   395→                            "left-0 right-1/2 rounded-r-full bg-m3-primary-container/44",
   396→                          bandKind === "mid" && "inset-x-0 bg-m3-primary-container/44",
   397→                          bandKind === "preview-end" &&
   398→                            "left-0 right-1/2 rounded-r-full bg-m3-primary-container/24",
   399→                          bandKind === "preview-mid" && "inset-x-0 bg-m3-primary-container/24"
   400→                        )}
   401→                      />
   402→                    )}
   403→                    <button
   404→                      type="button"
   405→                      data-iso={iso}
   406→                      disabled={disabled}
   407→                      tabIndex={iso === activeIso ? 0 : -1}
   408→                      aria-label={`${MONTH_NAMES[day.getMonth()]} ${day.getDate()}, ${day.getFullYear()}${
   409→                        isRangeStart ? ", start of range" : isRangeEnd ? ", end of range" : ""
   410→                      }`}
   411→                      onClick={() => handleSelect(day)}
   412→                      onKeyDown={(e) => handleDayKeyDown(e, idx)}
   413→                      {...rangeProps}
   414→                      className={cn(
   415→                        "m3-state m3-focus relative my-0.5 flex h-10 w-10 cursor-pointer items-center justify-center rounded-full md-body-large outline-none",
   416→                        disabled && "pointer-events-none opacity-38",
   417→                        !isSelected && isToday && "border border-m3-primary text-m3-primary",
   418→                        !isSelected && (previewEnd || previewRestart) && "border border-m3-primary",
   419→                        !isSelected &&
   420→                          !isToday &&
   421→                          !previewEnd &&
   422→                          !previewRestart &&
   423→                          (inMonth ? "text-m3-on-surface" : "text-m3-on-surface-variant")
   424→                      )}
   425→                    >
   426→                      {rangeOn ? (
   427→                        isRangeStart && (
   428→                          <motion.span
   429→                            layoutId={`${pillId}-start`}
   430→                            className={cn("absolute inset-0 rounded-full", selectedPillClass)}
   431→                            transition={springs.expressive}
   432→                          />
   433→                        )
   434→                      ) : (
   435→                        isSelected && (
   436→                          <motion.span
   437→                            layoutId={pillId}
   438→                            className={cn("absolute inset-0 rounded-full", selectedPillClass)}
   439→                            transition={springs.expressive}
   440→                          />
   441→                        )
   442→                      )}
   443→                      {rangeOn && isRangeEnd && !isRangeStart && (
   444→                        <motion.span
   445→                          layoutId={`${pillId}-end`}
   446→                          className={cn("absolute inset-0 rounded-full", selectedPillClass)}
   447→                          transition={springs.expressive}
   448→                        />
   449→                      )}
   450→                      <span
   451→                        className={cn(
   452→                          "relative z-10",
   453→                          (isSelected || isRangeStart || isRangeEnd) && selectedTextClass
   454→                        )}
   455→                      >
   456→                        {day.getDate()}
   457→                      </span>
   458→                    </button>
   459→                  </div>
   460→                );
   461→              })}
   462→            </div>
   463→          ))}
   464→        </div>
   465→      ) : (
   466→        /* Year grid: 4 columns, 1988 → current year + 10 */
   467→        <div className="m3-scroll grid h-[300px] grid-cols-4 content-start gap-1 overflow-y-auto pt-2">
   468→          {years.map((y) => {
   469→            const isCurrent = y === highlightYear;
   470→            return (
   471→              <button
   472→                type="button"
   473→                key={y}
   474→                ref={isCurrent ? selectedYearRef : undefined}
   475→                onClick={() => {
   476→                  setCursor(new Date(y, cursor.getMonth(), 1));
   477→                  setView("month");
   478→                }}
   479→                className={cn(
   480→                  "m3-state m3-focus md-body-large h-10 cursor-pointer rounded-full outline-none",
   481→                  isCurrent
   482→                    ? "bg-m3-primary-container text-m3-on-primary-container"
   483→                    : "text-m3-on-surface"
   484→                )}
   485→              >
   486→                {y}
   487→              </button>
   488→            );
   489→          })}
   490→        </div>
   491→      )}
   492→    </>
   493→  );
   494→}

   501→  value?: Date;
   502→  onChange?: (d: Date) => void;
   503→  minDate?: Date;
   504→  maxDate?: Date;
   505→  open?: boolean;
   506→  onOpenChange?: (open: boolean) => void;
   507→  closeOnSelect?: boolean;
   508→  className?: string;
   509→  selectionMode?: "single" | "range";
   510→  range?: DateRange;
   511→  onRangeChange?: (r: DateRange) => void;

   513→
   514→/**
   515→ * Official modal date picker: a role="dialog" on surface-container-high,
   516→ * 28dp corners, elevation 3, over a 32% scrim. Portrait (viewport < 600px)
   517→ * is 328×512dp with the selected-date header stacked on top (label-large
   518→ * supporting text + display-size headline + divider); landscape (≥ 600px)
   519→ * is 568×368dp with the header as a 168dp left column, vertically
   520→ * centered. Selection applies live (no action buttons) and closes the
   521→ * dialog when closeOnSelect; Escape and scrim tap always dismiss. Focus
   522→ * moves to the selected/today day on open, Tab is trapped, and focus is
   523→ * restored to the opener on close (same pattern as Dialog).
   524→ *
   525→ * selectionMode="range": the header shows "Start date" / "End date"
   526→ * placeholders (or the formatted pair "Aug 21 – Aug 28") and the dialog
   527→ * closes only once a COMPLETE range is picked — a start-only day tap keeps
   528→ * it open. Escape/scrim mid-selection dismiss without fabricating an end.
   529→ */
   530→const DatePickerModal = React.forwardRef<HTMLDivElement, DatePickerModalProps>(
   531→  function DatePickerModal(
   532→    {
   533→      value,
   534→      onChange,
   535→      minDate,
   536→      maxDate,
   537→      open = false,
   538→      onOpenChange,
   539→      closeOnSelect = true,
   540→      className,
   541→      selectionMode = "single",
   542→      range,
   543→      onRangeChange,
   544→    },
   545→    ref
   546→  ) {
   547→    const panelRef = React.useRef<HTMLDivElement | null>(null);
   548→    const restoreFocusRef = React.useRef<HTMLElement | null>(null);
   549→    // Official breakpoint: landscape layout at viewport ≥ 600dp (client-only)
   550→    const [landscape, setLandscape] = React.useState(false);
   551→    // Tracks picks when the caller renders uncontrolled, so the header stays live
   552→    const [picked, setPicked] = React.useState<Date | undefined>(undefined);
   553→    // Range-mode equivalent of `picked` for the uncontrolled header/live state
   554→    const [pickedRange, setPickedRange] = React.useState<DateRange>({});
   555→    const rangeOn = selectionMode === "range";
   556→    const rangeSel = rangeOn ? (range ?? pickedRange) : undefined;
   557→
   558→    React.useEffect(() => {
   559→      const mq = window.matchMedia("(min-width: 600px)");
   560→      const update = () => setLandscape(mq.matches);
   561→      update();
   562→      mq.addEventListener("change", update);
   563→      return () => mq.removeEventListener("change", update);
   564→    }, []);
   565→
   566→    // Escape dismiss + body scroll lock while open (Dialog pattern)
   567→    React.useEffect(() => {
   568→      if (!open) return;
   569→      const onKey = (e: KeyboardEvent) => {
   570→        if (e.key === "Escape") onOpenChange?.(false);
   571→      };
   572→      window.addEventListener("keydown", onKey);
   573→      const prevOverflow = document.body.style.overflow;
   574→      document.body.style.overflow = "hidden";
   575→      return () => {
   576→        window.removeEventListener("keydown", onKey);
   577→        document.body.style.overflow = prevOverflow;
   578→      };
   579→    }, [open, onOpenChange]);
   580→
   581→    // Initial focus on the selected/today day; restore focus to the opener on close
   582→    React.useEffect(() => {
   583→      if (!open) return;
   584→      restoreFocusRef.current = document.activeElement as HTMLElement | null;
   585→      const timer = window.setTimeout(() => {
   586→        const day = panelRef.current?.querySelector<HTMLButtonElement>(
   587→          'button[data-iso][tabindex="0"]'
   588→        );
   589→        if (day) day.focus();
   590→        else panelRef.current?.focus();
   591→      }, 0);
   592→      return () => {
   593→        window.clearTimeout(timer);
   594→        restoreFocusRef.current?.focus?.();
   595→      };
   596→    }, [open]);
   597→
   598→    // Trap Tab focus inside the dialog surface (Dialog pattern)
   599→    const handleTab = (e: React.KeyboardEvent<HTMLDivElement>) => {
   600→      if (e.key !== "Tab" || !panelRef.current) return;
   601→      const focusables = Array.from(
   602→        panelRef.current.querySelectorAll<HTMLElement>(FOCUSABLE)
   603→      ).filter((el) => !el.hasAttribute("disabled"));
   604→      if (focusables.length === 0) {
   605→        e.preventDefault();
   606→        return;
   607→      }
   608→      const first = focusables[0];
   609→      const last = focusables[focusables.length - 1];
   610→      const active = document.activeElement;
   611→      if (e.shiftKey && (active === first || active === panelRef.current)) {
   612→        e.preventDefault();
   613→        last.focus();
   614→      } else if (!e.shiftKey && active === last) {
   615→        e.preventDefault();
   616→        first.focus();
   617→      }
   618→    };
   619→
   620→    const handleSelect = (d: Date) => {
   621→      setPicked(d);
   622→      onChange?.(d);
   623→      // M3 live-apply: the value updates immediately; close when asked to
   624→      if (closeOnSelect) onOpenChange?.(false);
   625→    };
   626→
   627→    // Range picks arrive via the calendar's onRangeChange (range mode routes
   628→    // through it, not onChange). closeOnSelect range semantics: only a
   629→    // COMPLETE range closes the dialog — a start-only day tap keeps it open;
   630→    // Escape/scrim mid-selection dismiss without fabricating an end.
   631→    const handleRangeChange = (r: DateRange) => {
   632→      setPickedRange(r);
   633→      onRangeChange?.(r);
   634→      if (r.start && r.end && closeOnSelect) onOpenChange?.(false);
   635→    };
   636→
   637→    const headlineDate = value ?? picked ?? new Date();
   638→    const headline = formatHeadline(headlineDate);
   639→
   640→    // Range header: "Start date" / "End date" placeholders, or the formatted pair
   641→    let rangeHeadline: React.ReactNode;
   642→    if (rangeSel?.start && rangeSel?.end) {
   643→      rangeHeadline = `${formatShort(rangeSel.start)} – ${formatShort(rangeSel.end)}`;
   644→    } else if (rangeSel?.start) {
   645→      rangeHeadline = (
   646→        <>
   647→          {formatShort(rangeSel.start)}
   648→          {" – "}
   649→          <span className="text-m3-on-surface-variant">End date</span>
   650→        </>
   651→      );
   652→    } else {
   653→      rangeHeadline = <span className="text-m3-on-surface-variant">Start date</span>;
   654→    }
   655→    const headerLabel = rangeOn ? "Selected dates" : "Selected date";
   656→
   657→    return (
   658→      <AnimatePresence>
   659→        {open && (
   660→          <div className="fixed inset-0 z-[80] flex items-center justify-center p-4">
   661→            {/* 32% black scrim, click-to-dismiss */}
   662→            <motion.div
   663→              initial={{ opacity: 0 }}
   664→              animate={{ opacity: 1 }}
   665→              exit={{ opacity: 0 }}
   666→              // Named framer easing (tokens.ts easings.* are CSS strings);
   667→              // "easeOut" ≙ easings.standardDecelerate — same as Dialog scrim
   668→              transition={{
   669→                duration: durations.short4 / 1000,
   670→                ease: "easeOut",
   671→              }}
   672→              className="absolute inset-0 bg-m3-scrim/32"
   673→              onClick={() => onOpenChange?.(false)}
   674→            />
   675→            <motion.div
   676→              ref={panelRef}
   677→              role="dialog"
   678→              aria-modal="true"
   679→              aria-label={rangeOn ? "Choose date range" : "Choose date"}
   680→              tabIndex={-1}
   681→              onKeyDown={handleTab}
   682→              initial={{ scale: 0.9, opacity: 0 }}
   683→              animate={{ scale: 1, opacity: 1 }}
   684→              exit={{ scale: 0.9, opacity: 0 }}
   685→              transition={springs.expressive}
   686→              className={cn(
   687→                "m3-elevation-3 relative flex overflow-hidden rounded-[28px] bg-m3-surface-container-high outline-none",
   688→                landscape
   689→                  ? "h-[368px] max-h-[calc(100dvh-32px)] w-[568px] max-w-[calc(100vw-32px)]"
   690→                  : "h-[512px] max-h-[calc(100dvh-32px)] w-[328px] max-w-[calc(100vw-32px)] flex-col",
   691→                className
   692→              )}
   693→            >
   694→              {landscape ? (
   695→                <>
   696→                  {/* Header as a 168dp left column, vertically centered */}
   697→                  <div className="flex w-[168px] shrink-0 flex-col justify-center gap-1 px-4">
   698→                    <span className="md-label-large text-m3-on-surface-variant">
   699→                      {headerLabel}
   700→                    </span>
   701→                    {rangeOn ? (
   702→                      <span className="md-headline-small leading-tight text-m3-on-surface">
   703→                        {rangeHeadline}
   704→                      </span>
   705→                    ) : (
   706→                      <span className="md-headline-small leading-tight text-m3-on-surface">
   707→                        {headline}
   708→                      </span>
   709→                    )}
   710→                  </div>
   711→                  <div className="m3-scroll min-w-0 flex-1 overflow-y-auto px-4 py-1">
   712→                    <DatePickerCalendar
   713→                      value={value ?? picked}
   714→                      onChange={handleSelect}
   715→                      minDate={minDate}
   716→                      maxDate={maxDate}
   717→                      tone="primary-container"
   718→                      animatedHeader
   719→                      selectionMode={selectionMode}
   720→                      range={range}
   721→                      onRangeChange={rangeOn ? handleRangeChange : undefined}
   722→                    />
   723→                  </div>
   724→                </>
   725→              ) : (
   726→                <>
   727→                  {/* Header block on top: supporting text + display headline + divider */}
   728→                  <div className="flex shrink-0 flex-col gap-1 px-6 pb-3 pt-6">
   729→                    <span className="md-label-large text-m3-on-surface-variant">
   730→                      {headerLabel}
   731→                    </span>
   732→                    {rangeOn ? (
   733→                      <span className="md-headline-small leading-tight text-m3-on-surface">
   734→                        {rangeHeadline}
   735→                      </span>
   736→                    ) : (
   737→                      <span className="md-display-small text-m3-on-surface">{headline}</span>
   738→                    )}
   739→                  </div>
   740→                  <div className="h-px w-full shrink-0 bg-m3-outline-variant" />
   741→                  <div className="m3-scroll min-h-0 flex-1 overflow-y-auto px-4 pb-2 pt-2">
   742→                    <DatePickerCalendar
   743→                      value={value ?? picked}
   744→                      onChange={handleSelect}
   745→                      minDate={minDate}
   746→                      maxDate={maxDate}
   747→                      tone="primary-container"
   748→                      animatedHeader
   749→                      selectionMode={selectionMode}
   750→                      range={range}
   751→                      onRangeChange={rangeOn ? handleRangeChange : undefined}
   752→                    />
   753→                  </div>
   754→                </>
   755→              )}
   756→            </motion.div>
   757→          </div>
   758→        )}
   759→      </AnimatePresence>
   760→    );
   761→  }
   762→);
   763→
   764→DatePickerModal.displayName = "DatePickerModal";

   770→export interface DatePickerProps {
   771→  /** Currently selected date */
   772→  value?: Date;
   773→  onChange?: (d: Date) => void;
   774→  /** Earliest selectable date (days before are disabled) */
   775→  minDate?: Date;
   776→  /** Latest selectable date (days after are disabled) */
   777→  maxDate?: Date;
   778→  /**
   779→   * 'inline' (default): the embedded rounded-[28px] calendar panel.
   780→   * 'modal': the official M3 modal date picker — 328×512dp portrait /
   781→   * 568×368dp landscape dialog on surface-container-high (elevation 3,
   782→   * 28dp corners) over a 32% scrim, with a selected-date header
   783→   * ("Selected date" label-large + display-size headline). Controlled via
   784→   * open/onOpenChange like Dialog; selection applies live and no action
   785→   * buttons are shown.
   786→   */
   787→  presentation?: "inline" | "modal";
   788→  /** Modal only — controls visibility (fully controlled, Dialog/SearchView style). */
   789→  open?: boolean;
   790→  /** Modal only — called with the next open state on scrim click, Escape, or day pick. */
   791→  onOpenChange?: (open: boolean) => void;
   792→  /** Modal only — close automatically when a day is picked. Default true. */
   793→  closeOnSelect?: boolean;
   794→  /** Inline only: stretch to the container width */
   795→  fullWidth?: boolean;
   796→  /** 'single' (default) picks one date via value/onChange; 'range' picks a
   797→   * start/end pair via range/onRangeChange (tap start, then end; a tap
   798→   * before the start restarts — androidx DateRangePicker convention). */
   799→  selectionMode?: "single" | "range";
   800→  /** Range mode — controlled range; uncontrolled when omitted. */
   801→  range?: DateRange;
   802→  /** Range mode — fires on every tap with the next range (partial included). */
   803→  onRangeChange?: (r: DateRange) => void;
   804→  className?: string;
   805→}
   806→
   807→/**
   808→ * M3 Date Picker.
   809→ *
   810→ * Inline presentation — a rounded-[28px] surface-container-high panel
   811→ * with a month grid, a tappable "month year" header that flips into a
   812→ * 4-column year grid (1988 → current year + 10), today outlined in
   813→ * primary, and a spring-animated selection pill shared via layoutId
   814→ * (scoped per instance). The grid exposes ARIA grid roles with roving
   815→ * tabindex and arrow-key day navigation (←/→/↑/↓), Home/End for week
   816→ * start/end. Nav chevrons are 48dp targets.
   817→ *
   818→ * Modal presentation (presentation="modal") — the official M3 date
   819→ * picker dialog: 328×512dp portrait (header stacked on top of the
   820→ * calendar, divider between) / 568×368dp landscape (header as a 168dp
   821→ * vertically-centered left column) at viewport ≥ 600px, 32% scrim,
   822→ * spring scale 0.9→1 + fade entry (mirrored exit), live-applied
   823→ * selection with no action buttons, Escape/scrim dismissal, focus trap
   824→ * with initial focus on the selected/today day and restore to the
   825→ * opener, body scroll lock, and the same ARIA-grid calendar internals.
   826→ *
   827→ * selectionMode="range" — official M3 date-range selection in BOTH
   828→ * presentations: the first tap sets the start, a tap ≥ start completes the
   829→ * range, a tap < start (or any tap once complete) restarts; in-between
   830→ * days carry a continuous primary-container band (start cell = right half
   831→ * with rounded-left edge under the circle, end cell = left half with
   832→ * rounded-right edge, mid cells square, square cuts at week-row edges, a
   833→ * 4dp vertical inset keeps adjacent week stripes separate) while start/end
   834→ * days are 40dp circles; hovering/focusing while only the start is set
   835→ * previews the tentative range. ARIA: start/end/in-between days are
   836→ * aria-selected with "start/end of range" label suffixes; arrow-key
   837→ * navigation is unchanged. In the modal the header shows Start/End date
   838→ * placeholders (or the formatted pair) and closeOnSelect only closes once
   839→ * the range is complete — Escape/scrim mid-selection never fabricate an
   840→ * end. The forwardRef lands on the inline root / the modal dialog panel.
   841→ */
   842→export const DatePicker = React.forwardRef<HTMLDivElement, DatePickerProps>(function DatePicker(
   843→  {
   844→    value,
   845→    onChange,
   846→    minDate,
   847→    maxDate,
   848→    presentation = "inline",
   849→    open,
   850→    onOpenChange,
   851→    closeOnSelect = true,
   852→    fullWidth = false,
   853→    selectionMode = "single",
   854→    range,
   855→    onRangeChange,
   856→    className,
   857→  },
   858→  ref
   859→) {
   860→  if (presentation === "modal") {
   861→    return (
   862→      <DatePickerModal
   863→        ref={ref}
   864→        open={open}
   865→        onOpenChange={onOpenChange}
   866→        closeOnSelect={closeOnSelect}
   867→        value={value}
   868→        onChange={onChange}
   869→        minDate={minDate}
   870→        maxDate={maxDate}
   871→        selectionMode={selectionMode}
   872→        range={range}
   873→        onRangeChange={onRangeChange}
   874→        className={className}
   875→      />
   876→    );
   877→  }
   878→
   879→  return (
   880→    <div
   881→      ref={ref}
   882→      className={cn(
   883→        "rounded-[28px] bg-m3-surface-container-high p-6",
   884→        fullWidth ? "w-full" : "w-[328px]",
   885→        className
   886→      )}
   887→    >
   888→      <DatePickerCalendar
   889→        value={value}
   890→        onChange={onChange}
   891→        minDate={minDate}
   892→        maxDate={maxDate}
   893→        selectionMode={selectionMode}
   894→        range={range}
   895→        onRangeChange={onRangeChange}
   896→      />
   897→    </div>
   898→  );
   899→});
